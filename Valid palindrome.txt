class Solution {
    public boolean isPalindrome(String s) {

        String r = "";

        for(int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);

            if(Character.isLetterOrDigit(ch)) {
                r = r + ch;
            }
        }

        String d = "";

        for(int i = r.length() - 1; i >= 0; i--) {
            d = d + r.charAt(i);
        }

        return r.equalsIgnoreCase(d);
    }
}
